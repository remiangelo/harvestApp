<CodeBlock language="typescript">
// lib/swipes.ts - Enhanced swipe tracking
import { supabase } from './supabase';

export interface SwipeResult {
  success: boolean;
  isMatch: boolean;
  error?: string;
}

export async function saveSwipe(
  userId: string,
  swipedUserId: string,
  action: 'like' | 'nope' | 'super_like'
): Promise<SwipeResult> {
  try {
    // Save the swipe
    const { error: swipeError } = await supabase
      .from('swipes')
      .insert({
        user_id: userId,
        swiped_user_id: swipedUserId,
        action
      });

    if (swipeError) throw swipeError;

    // Check for mutual match if it's a like/super_like
    if (action === 'like' || action === 'super_like') {
      const { data: mutualSwipe } = await supabase
        .from('swipes')
        .select('action')
        .eq('user_id', swipedUserId)
        .eq('swiped_user_id', userId)
        .in('action', ['like', 'super_like'])
        .single();

      if (mutualSwipe) {
        // Create match
        await createMatch(userId, swipedUserId);
        return { success: true, isMatch: true };
      }
    }

    return { success: true, isMatch: false };

} catch (error) {
console.error('Error saving swipe:', error);
return {
success: false,
isMatch: false,
error: error.message
};
}
}

// Get profiles user hasn't swiped on
export async function getUnswipedProfiles(userId: string) {
const { data: swipedUserIds } = await supabase
.from('swipes')
.select('swiped_user_id')
.eq('user_id', userId);

const swipedIds = swipedUserIds?.map(s => s.swiped_user_id) || [];

const { data: profiles } = await supabase
.from('profiles')
.select('\*')
.neq('id', userId)
.not('id', 'in', `(${swipedIds.join(',')})`)
.limit(50);

return profiles || [];
}

// lib/matches.ts
import { supabase } from './supabase';

export async function createMatch(userId1: string, userId2: string) {
  // Ensure consistent ordering
  const [user1_id, user2_id] = userId1 < userId2 
    ? [userId1, userId2] 
    : [userId2, userId1];

try {
const { data, error } = await supabase
.from('matches')
.insert({
user1_id,
user2_id
})
.select()
.single();

    if (error) throw error;

    // Send push notification (implement later)
    // await sendMatchNotification(userId1, userId2);

    return { success: true, match: data };

} catch (error) {
console.error('Error creating match:', error);
return { success: false, error: error.message };
}
}

export async function getUserMatches(userId: string) {
  const { data, error } = await supabase
    .from('matches')
    .select(`
      *,
      user1:user1_id(id, nickname, photos, bio),
      user2:user2_id(id, nickname, photos, bio)
    `)
    .or(`user1_id.eq.${userId},user2_id.eq.${userId}`)
    .eq('is_active', true)
    .order('created_at', { ascending: false });

if (error) throw error;

// Format matches to always have the other user as 'match'
return data?.map(match => ({
...match,
match: match.user1_id === userId ? match.user2 : match.user1,
unreadCount: match.user1_id === userId
? match.user1_unread_count
: match.user2_unread_count
}));
}

  match_id UUID REFERENCES matches(id) ON DELETE CASCADE,
  sender_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,

  message_type VARCHAR(20) DEFAULT 'text' 
    CHECK (message_type IN ('text', 'image', 'gif', 'voice'))
);

-- Indexes
CREATE INDEX idx_messages_match_id ON messages(match_id);
CREATE INDEX idx_messages_sender_id ON messages(sender_id);
CREATE INDEX idx_messages_created_at ON messages(created_at);

-- RLS Policies
ALTER TABLE messages ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users can view messages in their matches"
ON messages FOR SELECT
USING (
EXISTS (
SELECT 1 FROM matches
WHERE matches.id = messages.match_id
AND (matches.user1_id = auth.uid() OR matches.user2_id = auth.uid())
)
);

CREATE POLICY "Users can send messages to their matches"
ON messages FOR INSERT
WITH CHECK (
sender_id = auth.uid() AND
EXISTS (
SELECT 1 FROM matches
WHERE matches.id = match_id
AND (matches.user1_id = auth.uid() OR matches.user2_id = auth.uid())
)
);

#### 2. Real-time Chat Implementation

<CodeBlock language="typescript">
// lib/chat.ts
import { supabase } from './supabase';
import { RealtimeChannel } from '@supabase/supabase-js';

export class ChatService {
  private channel: RealtimeChannel | null = null;

async sendMessage(matchId: string, senderId: string, content: string) {
const { data, error } = await supabase
.from('messages')
.insert({
match_id: matchId,
sender_id: senderId,
content
})
.select()
.single();

    if (error) throw error;

    // Update last message timestamp
    await supabase
      .from('matches')
      .update({ last_message_at: new Date().toISOString() })
      .eq('id', matchId);

    return data;

}

subscribeToMessages(matchId: string, onMessage: (message: any) => void) {
this.channel = supabase
.channel(`messages:${matchId}`)
.on(
'postgres_changes',
{
event: 'INSERT',
schema: 'public',
table: 'messages',
filter: `match_id=eq.${matchId}`
},
(payload) => {
onMessage(payload.new);
}
)
.subscribe();

    return () => {
      if (this.channel) {
        supabase.removeChannel(this.channel);
      }
    };

}

async getMessages(matchId: string, limit = 50) {
const { data, error } = await supabase
.from('messages')
.select('\*, sender:sender_id(id, nickname, photos)')
.eq('match_id', matchId)
.order('created_at', { ascending: false })
.limit(limit);

    if (error) throw error;
    return data?.reverse() || [];

}

async markAsRead(matchId: string, userId: string) {
await supabase
.from('messages')
.update({ read_at: new Date().toISOString() })
.eq('match_id', matchId)
.neq('sender_id', userId)
.is('read_at', null);
}
}

export const chatService = new ChatService();
</CodeBlock>

// lib/notifications.ts
import * as Notifications from 'expo-notifications';
import * as Device from 'expo-device';
import { Platform } from 'react-native';
import { supabase } from './supabase';

Notifications.setNotificationHandler({
handleNotification: async () => ({
shouldShowAlert: true,
shouldPlaySound: true,
shouldSetBadge: true,
}),
});

export class NotificationService {
  async registerForPushNotifications() {
    if (!Device.isDevice) {
      console.log('Must use physical device for Push Notifications');
      return;
    }

    const { status: existingStatus } = await Notifications.getPermissionsAsync();
    let finalStatus = existingStatus;

    if (existingStatus !== 'granted') {
      const { status } = await Notifications.requestPermissionsAsync();
      finalStatus = status;
    }

    if (finalStatus !== 'granted') {
      console.log('Failed to get push token for push notification!');
      return;
    }

    const token = (await Notifications.getExpoPushTokenAsync()).data;

    // Save token to user profile
    await this.saveTokenToDatabase(token);

    if (Platform.OS === 'android') {
      Notifications.setNotificationChannelAsync('default', {
        name: 'default',
        importance: Notifications.AndroidImportance.MAX,
        vibrationPattern: [0, 250, 250, 250],
        lightColor: '#FF231F7C',
      });
    }

    return token;

}

const { error } = await supabase
.from('profiles')
